// فتح الواجهة الرئيسية عند طلب الرابط (doGet)
function doGet() {
  return HtmlService.createHtmlOutputFromFile('Index')
    .setTitle('نظام إدارة مؤسسة حسن خالد حسن فران للمقاولات')
    .setXframeOptionsMode(HtmlService.XFrameOptionsMode.ALLOWALL)
    .addMetaTag('viewport', 'width=device-width, initial-scale=1.0');
}

// المعالجة المباشرة لطلبات HTTP POST (لتسجيل الدخول المباشر عبر API)
function doPost(e) {
  try {
    const contents = JSON.parse(e.postData.contents);
    if (contents.action === 'login') {
      const email = contents.email;
      const password = contents.password;
      
      const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName('المستخدمين');
      if (!sheet) {
        return createJsonResponse({ success: false, message: 'جدول المستخدمين غير موجود' });
      }
      
      const data = sheet.getDataRange().getValues();
      const headers = data[0];
      const emailIdx = headers.indexOf('البريد الإلكتروني');
      const passIdx = headers.indexOf('كلمة المرور');
      const nameIdx = headers.indexOf('الاسم');
      const roleIdx = headers.indexOf('الدور');

      for (let i = 1; i < data.length; i++) {
        if (data[i][emailIdx] == email && data[i][passIdx] == password) {
          const user = {
            id: data[i][0],
            name: data[i][nameIdx],
            email: data[i][emailIdx],
            role: data[i][roleIdx]
          };
          return createJsonResponse({ success: true, user: user });
        }
      }
      return createJsonResponse({ success: false, message: 'اسم المستخدم أو كلمة المرور غير صحيحة' });
    }
  } catch (err) {
    return createJsonResponse({ success: false, message: err.toString() });
  }
}

function createJsonResponse(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}

// --- التعامل مع البيانات (CRUD) ---

// جلب بيانات جدول محدد كـ Objects
function getData(sheetName) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  if (!sheet) return [];
  
  const data = sheet.getDataRange().getValues();
  if (data.length <= 1) return [];

  const headers = data[0];
  const result = [];

  for (let i = 1; i < data.length; i++) {
    let rowObj = {};
    for (let j = 0; j < headers.length; j++) {
      rowObj[headers[j]] = data[i][j];
    }
    result.push(rowObj);
  }
  return result;
}

// حفظ أو تعديل سجل في جدول محدد
function saveRecord(sheetName, record) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  if (!sheet) return { success: false, message: 'الجدول غير موجود' };

  const data = sheet.getDataRange().getValues();
  const headers = data[0];

  // إذا لم يكن هناك ID، قم بإنشاء جديد
  if (!record['ID']) {
    record['ID'] = new Date().getTime().toString();
    const newRow = headers.map(h => record[h] !== undefined ? record[h] : '');
    sheet.appendRow(newRow);
    return { success: true, action: 'added', id: record['ID'] };
  } else {
    // البحث عن السجل والتعديل عليه
    const idIdx = headers.indexOf('ID');
    for (let i = 1; i < data.length; i++) {
      if (data[i][idIdx] == record['ID']) {
        const rowNum = i + 1;
        headers.forEach((h, j) => {
          if (record[h] !== undefined) {
            sheet.getRange(rowNum, j + 1).setValue(record[h]);
          }
        });
        return { success: true, action: 'updated' };
      }
    }
  }
}

// حذف سجل حسب الـ ID
function deleteRecord(sheetName, id) {
  const sheet = SpreadsheetApp.getActiveSpreadsheet().getSheetByName(sheetName);
  if (!sheet) return { success: false };

  const data = sheet.getDataRange().getValues();
  const headers = data[0];
  const idIdx = headers.indexOf('ID');

  for (let i = 1; i < data.length; i++) {
    if (data[i][idIdx] == id) {
      sheet.deleteRow(i + 1);
      return { success: true };
    }
  }
  return { success: false };
}

// جلب إحصائيات لوحة التحكم والانتهاءات
function getDashboardData() {
  const projects = getData('المشاريع');
  const expenses = getData('المصروفات');
  const orgDocs = getData('وثائق المؤسسة');
  const employees = getData('الموظفين');

  let totalContractValue = 0;
  let totalCollected = 0;

  projects.forEach(p => {
    totalContractValue += parseFloat(p['القيمة الاجمالية']) || 0;
    totalCollected += parseFloat(p['المستخلص المحصل']) || 0;
  });

  let totalExpenses = 0;
  expenses.forEach(e => {
    totalExpenses += parseFloat(e['المبلغ']) || 0;
  });

  const totalRemaining = totalContractValue - totalCollected;

  // حساب التنبيهات القريبة من الانتهاء (أقل من 30 يوم)
  const today = new Date();

  const expiringDocs = orgDocs.filter(d => {
    if (!d['تاريخ الانتهاء']) return false;
    const exp = new Date(d['تاريخ الانتهاء']);
    const diffDays = (exp - today) / (1000 * 60 * 60 * 24);
    return diffDays <= 30;
  });

  const expiringIqamas = employees.filter(emp => {
    if (!emp['تاريخ انتهاء الإقامة']) return false;
    const exp = new Date(emp['تاريخ انتهاء الإقامة']);
    const diffDays = (exp - today) / (1000 * 60 * 60 * 24);
    return diffDays <= 30;
  });

  return {
    stats: {
      totalContractValue: totalContractValue,
      totalCollected: totalCollected,
      totalRemaining: totalRemaining,
      totalExpenses: totalExpenses
    },
    expiringDocs: expiringDocs,
    expiringIqamas: expiringIqamas
  };
}

// رفع الملفات إلى Google Drive
function uploadFileToDrive(fileObj) {
  try {
    const folderName = 'مرفقات_النظام';
    let folder;
    const folders = DriveApp.getFoldersByName(folderName);
    
    if (folders.hasNext()) {
      folder = folders.next();
    } else {
      folder = DriveApp.createFolder(folderName);
    }

    const contentType = fileObj.mimeType;
    const base64Data = fileObj.data.split(',')[1];
    const decodedData = Utilities.base64Decode(base64Data);
    const blob = Utilities.newBlob(decodedData, contentType, fileObj.fileName);

    const file = folder.createFile(blob);
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);

    return { success: true, url: file.getUrl() };
  } catch (err) {
    return { success: false, message: err.toString() };
  }
}
